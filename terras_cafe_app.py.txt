import tkinter as tk
from tkinter import ttk, messagebox
import sqlite3

# ==========================
# CONEXÃO COM O BANCO
# ==========================

con = sqlite3.connect("terras_cafe.db")
cursor = con.cursor()

cursor.execute("""
CREATE TABLE IF NOT EXISTS tb_clientes(
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    telefone TEXT NOT NULL,
    email TEXT,
    aniversario TEXT,
    produto TEXT
)
""")

con.commit()

# ==========================
# JANELA PRINCIPAL
# ==========================

janela = tk.Tk()
janela.title("☕ Terra's Café - Sistema de Clientes")
janela.geometry("1050x650")
janela.configure(bg="#f7f2ea")

fonte = ("Arial",11)

frame_form = tk.Frame(janela,bg="#f7f2ea")
frame_form.pack(side="left",fill="y",padx=15,pady=15)

tk.Label(frame_form,text="Nome",bg="#f7f2ea",font=fonte).pack(anchor="w")
nome = tk.Entry(frame_form,width=35)
nome.pack()

tk.Label(frame_form,text="Telefone",bg="#f7f2ea",font=fonte).pack(anchor="w")
telefone = tk.Entry(frame_form,width=35)
telefone.pack()

tk.Label(frame_form,text="E-mail",bg="#f7f2ea",font=fonte).pack(anchor="w")
email = tk.Entry(frame_form,width=35)
email.pack()

tk.Label(frame_form,text="Aniversário",bg="#f7f2ea",font=fonte).pack(anchor="w")
aniversario = tk.Entry(frame_form,width=35)
aniversario.pack()

tk.Label(frame_form,text="Produto Favorito",bg="#f7f2ea",font=fonte).pack(anchor="w")
produto = tk.Entry(frame_form,width=35)
produto.pack()

# ==========================
# TABELA
# ==========================

frame_lista = tk.Frame(janela)
frame_lista.pack(fill="both",expand=True,padx=10,pady=10)

colunas=("ID","Nome","Telefone","Email","Aniversário","Produto")

tabela=ttk.Treeview(frame_lista,columns=colunas,show="headings")

for c in colunas:
    tabela.heading(c,text=c)
    tabela.column(c,width=150)

tabela.pack(fill="both",expand=True)

# ==========================
# FUNÇÕES
# ==========================

def limpar():
    nome.delete(0,tk.END)
    telefone.delete(0,tk.END)
    email.delete(0,tk.END)
    aniversario.delete(0,tk.END)
    produto.delete(0,tk.END)

def carregar():
    tabela.delete(*tabela.get_children())

    cursor.execute("SELECT * FROM tb_clientes")

    for linha in cursor.fetchall():
        tabela.insert("",tk.END,values=linha)

carregar()
# ==========================
# CADASTRAR CLIENTE
# ==========================

def salvar():

    if nome.get() == "" or telefone.get() == "":
        messagebox.showwarning(
            "Aviso",
            "Nome e telefone são obrigatórios!"
        )
        return

    cursor.execute("""
    INSERT INTO tb_clientes
    (nome, telefone, email, aniversario, produto)

    VALUES(?,?,?,?,?)
    """,(

        nome.get(),
        telefone.get(),
        email.get(),
        aniversario.get(),
        produto.get()

    ))

    con.commit()

    carregar()
    limpar()

    messagebox.showinfo(
        "Sucesso",
        "Cliente cadastrado!"
    )


# ==========================
# SELECIONAR CLIENTE
# ==========================

id_selecionado = None

def selecionar(event):

    global id_selecionado

    item = tabela.focus()

    if item == "":
        return

    dados = tabela.item(item)

    valores = dados["values"]

    if len(valores) == 0:
        return

    id_selecionado = valores[0]

    limpar()

    nome.insert(0,valores[1])
    telefone.insert(0,valores[2])
    email.insert(0,valores[3])
    aniversario.insert(0,valores[4])
    produto.insert(0,valores[5])


tabela.bind("<<TreeviewSelect>>", selecionar)


# ==========================
# EDITAR CLIENTE
# ==========================

def editar():

    global id_selecionado

    if id_selecionado is None:
        messagebox.showwarning(
            "Aviso",
            "Selecione um cliente."
        )
        return

    cursor.execute("""

    UPDATE tb_clientes

    SET

    nome=?,
    telefone=?,
    email=?,
    aniversario=?,
    produto=?

    WHERE id=?

    """,(

        nome.get(),
        telefone.get(),
        email.get(),
        aniversario.get(),
        produto.get(),
        id_selecionado

    ))

    con.commit()

    carregar()

    limpar()

    id_selecionado = None

    messagebox.showinfo(
        "Sucesso",
        "Registro atualizado!"
    )


# ==========================
# EXCLUIR CLIENTE
# ==========================

def excluir():

    global id_selecionado

    if id_selecionado is None:
        messagebox.showwarning(
            "Aviso",
            "Selecione um cliente."
        )
        return

    op = messagebox.askyesno(
        "Excluir",
        "Deseja realmente excluir?"
    )

    if op:

        cursor.execute(
            "DELETE FROM tb_clientes WHERE id=?",
            (id_selecionado,)
        )

        con.commit()

        carregar()

        limpar()

        id_selecionado = None

        messagebox.showinfo(
            "Sucesso",
            "Cliente removido."
        )
# ==========================
# PESQUISAR CLIENTE
# ==========================

tk.Label(
    frame_form,
    text="Pesquisar Nome",
    bg="#f7f2ea",
    font=fonte
).pack(pady=(15,0))

pesquisa = tk.Entry(frame_form, width=35)
pesquisa.pack()


def pesquisar():

    tabela.delete(*tabela.get_children())

    cursor.execute("""

    SELECT * FROM tb_clientes

    WHERE nome LIKE ?

    """,('%'+pesquisa.get()+'%',))

    for linha in cursor.fetchall():

        tabela.insert("",tk.END,values=linha)


tk.Button(
    frame_form,
    text="Pesquisar",
    command=pesquisar,
    width=30,
    bg="#6f4e37",
    fg="white"
).pack(pady=5)


# ==========================
# BOTÕES
# ==========================

tk.Button(
    frame_form,
    text="Salvar",
    command=salvar,
    width=30,
    bg="green",
    fg="white"
).pack(pady=3)

tk.Button(
    frame_form,
    text="Editar",
    command=editar,
    width=30,
    bg="orange",
    fg="white"
).pack(pady=3)

tk.Button(
    frame_form,
    text="Excluir",
    command=excluir,
    width=30,
    bg="red",
    fg="white"
).pack(pady=3)

tk.Button(
    frame_form,
    text="Limpar",
    command=limpar,
    width=30
).pack(pady=3)

tk.Button(
    frame_form,
    text="Atualizar Lista",
    command=carregar,
    width=30
).pack(pady=10)


# ==========================
# FECHAR APLICAÇÃO
# ==========================

def fechar():

    con.close()

    janela.destroy()


janela.protocol("WM_DELETE_WINDOW", fechar)

carregar()

janela.mainloop()
